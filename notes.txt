### `docker-compose.yml` (simplified)

```
services:
  web:
    build: ./docker/web
    ports: ["3000:3000"]
    depends_on: [api]

  api:
    build: ./docker/api
    ports: ["4000:4000"]
    depends_on: [postgres]

  postgres:
    image: postgres:15
    volumes:
      - pgdata:/var/lib/postgresql/data

  nginx:
    image: nginx
    ports: ["80:80"]
```
